CREATE TABLE IF NOT EXISTS `time` (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(255) NOT NULL,
    sigla CHAR(3) NOT NULL,
    qtd_copas INT DEFAULT 0,
    ranking_fifa DECIMAL(10, 2),
    ativo BOOLEAN DEFAULT TRUE,
    data_fundacao DATE
);

CREATE TABLE IF NOT EXISTS jogo (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_time_mandante INT NOT NULL,
    id_time_visitante INT NOT NULL,
    nr_gols_mandante INT DEFAULT 0,
    nr_gols_visitante INT DEFAULT 0,
    publico_pagante INT,
    renda_total DECIMAL(15, 2),
    realizado BOOLEAN DEFAULT FALSE,
    data_jogo DATETIME,
    
    CONSTRAINT fk_mandante FOREIGN KEY (id_time_mandante) REFERENCES `time`(id),
    CONSTRAINT fk_visitante FOREIGN KEY (id_time_visitante) REFERENCES `time`(id)
);
